      `def _active_trees(config: dict) -> list[dict]:`
        return [t for t in config.get("trees", []) if t.get("active", True)].
      `def spacing_penalty(config: dict, min_spacing_m: float = MIN_SPACING_M) -> float:` (RULES-01)
        For each unordered pair of active trees compute Euclidean distance in (x_m, y_m).
        For pairs closer than min_spacing_m add `(min_spacing_m - dist) / min_spacing_m` to the
        penalty (clamped >= 0). Pairs at/above min_spacing add 0. Return total penalty (>= 0.0).
        0 or 1 trees → 0.0.
      `def species_diversity_score(config: dict) -> float:` (RULES-02)
        Count species among active trees. Shannon index H = -sum(p_i * ln(p_i)). Normalize by
        ln(n_distinct) so a balanced mix → 1.0 and a monoculture → 0.0. < 2 trees → 0.0.
        Use math.log; guard log(0). Fall back to simple distinct-count ratio if only 1 species.
      `def ecological_score(config: dict) -> float:`
        Normalize spacing_penalty by active tree count (penalty / max(1, n_pairs)) → in [0,1+];
        coherence = 0.5 * (1 - min(1.0, norm_penalty)) + 0.5 * species_diversity_score(config).
        Return round(coherence, 4), clamped to [0.0, 1.0].
