<objective>
Replace the flat €350 CapEx / €35-OpEx placeholders in `cost_model.py` with an itemized, fully-loaded lifecycle cost table (COST-03 / D-01..D-04). Each of the six named line items carries its own `value`, `unit`, `source`, and `confidence` tag so the audit trail shows exactly what is grounded versus estimated. # New in coolspend/cost_model.py

```python
from dataclasses import dataclass, field

# Confidence tags already used across the codebase:
VERIFIED = "VERIFIED"   # cited, independently confirmed
DECLARED = "DECLARED"   # named source anchor + reasonable magnitude, not locally verified
PENDING  = "PENDING"    # source identified but unconfirmed / paywalled / region-mismatch

@dataclass(frozen=True)
class CostLine:
    """One itemized lifecycle cost line."""
    key: str            # machine key, e.g. "tree_stock"
    label: str          # human label, e.g. "Tree stock (large-caliper nursery)"
    value: float        # EUR (CapEx lines) or EUR/yr (opex line)
    unit: str           # "EUR/tree" | "EUR/tree/yr"
    kind: str           # "capex" | "opex"
    source: str         # named source anchor — NO fabricated DOIs
    confidence: str     # VERIFIED | DECLARED | PENDING

@dataclass
class CostTable:
    """Editable per-tree cost table (COST-04 loads/edits this)."""
    lines: list[CostLine]
    label: str = "illustrative European mid-range — verify locally"

    def capex_total(self) -> float: ...          # sum of capex line values
    def opex_per_year(self) -> float: ...         # sum of opex line values
    def per_tree_cost(self, horizon_years: int) -> float: ...  # capex + opex*horizon

# Module-level default itemization (the shipped defaults):
DEFAULT_COST_TABLE: CostTable
```
